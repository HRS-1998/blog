## 使用

![](http://doc.uc108.org:8002/server/../Public/Uploads/2022-12-23/63a518de9b5ca.png)

```html
<div id="container"></div>
<!-- 加载地图JSAPI脚本 -->
<script src="https://webapi.amap.com/maps?v=1.4.15&key=1d67e57913135953a0e086962c8ccf1f&&plugin=AMap.ToolBar"></script>
<script>
  var map = new AMap.Map('container', {
    resizeEnable: true, //是否监控地图容器尺寸变化
    zoom: 16, //初始化地图层级
    center: [120.202834, 30.15471], //初始化地图中心点
  });
  var toolBar = new AMap.ToolBar({
    visible: true,
  });
  map.addControl(toolBar);
  addMarker();

  //添加marker标记
  function addMarker() {
    map.clearMap();

    //构建自定义信息窗体
    var infoWindow = new AMap.InfoWindow({
      offset: new AMap.Pixel(0, -33),
      anchor: 'bottom-center',
      content:
        '<div style="font-size:14px;line-height:28px;"><div style="font-weight: 700;color:#CC5522">海云科技园</div><div>地址：浙江省杭州市滨江区塘子堰路199号</div></div>',
    });
    var marker = new AMap.Marker({
      map: map,
      position: [120.202834, 30.15471],
    });
    //鼠标点击marker弹出自定义的信息窗体
    AMap.event.addListener(marker, 'click', function () {
      infoWindow.open(map, marker.getPosition());
    });
    infoWindow.open(map, marker.getPosition());
  }
</script>
```
